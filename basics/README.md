0	pwd	Affiche le chemin complet du dossier actuel.
1	ls	Liste le contenu du répertoire courant.
2	cd	Change de répertoire pour revenir dans le dossier personnel de l'utilisateur.
3	ls -l	Affiche le contenu du dossier en format liste détaillée.
4	ls -la	Affiche tous les fichiers, y compris les fichiers cachés.
5	ls -n	Liste les fichiers en affichant les ID numériques (UID/GID) au lieu des noms d'utilisateur.
6	mkdir	Crée un nouveau répertoire (make kirectory).
7	mv	Déplace un fichier d'un endroit à un autre (ou le renomme).
8	rm	Supprime définitivement un fichier (remove).
9	rmdir	Supprime un répertoire, à condition qu'il soit vide.
10	cd -	Permet de revenir instantanément au répertoire précédent.
11	ls -aR	Liste les fichiers de manière récursive (inclut tous les sous-dossiers).
12	file	Analyse et affiche le type de contenu d'un fichier .
13	ln -s	Crée un lien symbolique (un raccourci) vers un fichier.
14	cp -u	Copie des fichiers mais uniquement s'ils sont plus récents que la version existante.
15	mv [A-Z]*	Déplace tous les fichiers commençant par une majuscule.
16	rm *~	Supprime tous les fichiers se terminant par ~ .
17	mkdir -p	Crée une structure de répertoires imbriqués en créant les parents si besoin.
